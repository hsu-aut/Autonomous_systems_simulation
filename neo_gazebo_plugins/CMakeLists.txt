cmake_minimum_required(VERSION 3.8)
project(neo_gazebo_plugins)
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)

add_library(neo_planar_move SHARED src/neo_planar_move.cpp)
target_include_directories(neo_planar_move PUBLIC include ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(neo_planar_move ${GAZEBO_LIBRARIES})
ament_target_dependencies(neo_planar_move rclcpp gazebo_ros gazebo_dev geometry_msgs nav_msgs tf2_geometry_msgs tf2_ros)

install(TARGETS neo_planar_move LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_package()
